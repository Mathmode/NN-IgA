\documentclass[border=6pt]{standalone}
\usepackage{pgfplots}
\usepackage{amsmath}
\usepgfplotslibrary{groupplots}
\pgfplotsset{compat=1.18}

\newlength{\plotsize}
\setlength{\plotsize}{5.0cm}

\pgfplotsset{
  gradheat/.style={
    matrix plot*,
    mesh/cols=81,
    point meta=explicit,
    shader=interp
  },
  knots/.style={
    black,
    line width=0.3pt,
    mark=none,
    forget plot
  },
  corner/.style={
    black,
    line width=0.9pt,
    mark=none,
    forget plot
  },
  rowcolorbar/.style={
    colorbar right,
    colorbar style={
      scale only axis,
      height=\plotsize,
      width=2.6mm,
      % log10 colour scale: ticks placed at log10(value), labelled with the value
      ytick={-1,-0.522879,0,0.477121,0.69897},
      yticklabels={$0.1$,$0.3$,$1$,$3$,$5$},
      yticklabel style={font=\scriptsize},
      tick label style={font=\scriptsize},
      scaled y ticks=false,
      ylabel={$|\nabla u_h|$},
      ylabel style={font=\scriptsize}
    }
  },
  colormap={gradcividis}{
    rgb255(0cm)=(0,32,76);
    rgb255(1cm)=(40,72,122);
    rgb255(2cm)=(87,116,137);
    rgb255(3cm)=(142,159,129);
    rgb255(4cm)=(208,210,98)
  }
}

% blank the removed quadrant within the zoom box + trace the re-entrant corner
\newcommand{\Lblankzoom}{%
  \fill[white] (axis cs:0.5,0.25) rectangle (axis cs:0.75,0.5);
  \addplot[corner] coordinates {(0.5,0.25) (0.5,0.5) (0.75,0.5)};
}
