\newlength{\plotsize}
\setlength{\plotsize}{0.27\textwidth}

\pgfplotsset{
  heat/.style={
    matrix plot*,
    mesh/cols=81,
    point meta=explicit,
    shader=interp
  },
  meshov/.style={
    black,
    line width=0.25pt,
    mark=none,
    forget plot
  },
  lshape/.style={
    black,
    line width=0.8pt,
    mark=none,
    forget plot
  },
  rowcolorbar/.style={
    colorbar right,
    colorbar style={
      scale only axis,
      height=\plotsize,
      width=2.5mm,
      scaled y ticks=base 10:2,
      ytick scale label code/.code={},
      title={$\cdot 10^{-2}$},
      title style={
        at={(axis description cs:0.5,1.04)},
        anchor=south,
        font=\scriptsize,
        inner sep=0pt
      },
      yticklabel style={font=\scriptsize},
      tick label style={font=\scriptsize}
    }
  },
  colormap={surfaceblue}{
    rgb255(0cm)=(35,55,120);
    rgb255(1cm)=(70,120,180);
    rgb255(2cm)=(120,190,210);
    rgb255(3cm)=(230,230,170);
    rgb255(4cm)=(180,70,50)
  }
}
