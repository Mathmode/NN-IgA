\documentclass[border=6pt]{standalone}

\newlength{\plotsize}

  heat/.style={
    matrix plot*,
    mesh/cols=80,
    point meta=explicit,
    shader=interp
  },

    line width=0.25pt,

      height=\plotsize,

      yticklabel style={font=\scriptsize},
      tick label style={font=\scriptsize}
    }
  },
  colormap={surfaceblue}{
    rgb255(0cm)=(35,55,120);
    rgb255(1cm)=(70,120,180);
    rgb255(2cm)=(120,190,210);
    rgb255(3cm)=(230,230,170);
    rgb255(4cm)=(180,70,50)
  }
}
